trigger operators_delete_revokes_access: CREATE TRIGGER operators_delete_revokes_access BEFORE DELETE ON operators BEGIN UPDATE api_keys SET revoked_at = CAST(strftime('%s', 'now') AS INTEGER) * 1000 WHERE created_by = OLD.id AND revoked_at IS NULL; UPDATE oauth_tokens SET revoked_at = CAST(strftime('%s', 'now') AS INTEGER) * 1000 WHERE revoked_at IS NULL AND grant_id IN (SELECT id FROM oauth_grants WHERE operator_id = OLD.id); UPDATE oauth_grants SET revoked_at = CAST(strftime('%s', 'now') AS INTEGER) * 1000 WHERE operator_id = OLD.id AND revoked_at IS NULL; END
